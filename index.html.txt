<script>
const msgs=document.getElementById('msgs');
const qInput=document.getElementById('qInput');
const sendBtn=document.getElementById('sendBtn');
const kb=[
{q:"કપાસમાં aphid ની સારવાર કેવી રીતે કરવી?", a:"Neem અથવા insecticidal soap નો ઉપયોગ કરો. મોટા infestation માટે insecticide વાપરો."},
{q:"धान ક્યારે વાવવું?", a:"પ્રદેશ મુજબ જુઓ, સામાન્ય રીતે મોનસૂન પહેલાં વાવવું યોગ્ય છે."},
{q:"ગાંધો માટે ખાતર કેટલું આપવું?", a:"80-120 kg/ha N split doses માં આપવી."}
];
function addMsg(who,text){
const div=document.createElement('div');div.className='msg '+who;div.innerText=text;
msgs.appendChild(div);msgs.scrollTop=msgs.scrollHeight;
}
sendBtn.addEventListener('click',()=>{
const q=qInput.value.trim(); if(!q) return;
addMsg('user',q);
let found=false;
for(const rec of kb){
if(q.includes(rec.q.split(' ')[0])){ addMsg('bot',rec.a); found=true; break; }
}
if(!found) addMsg('bot','માફ કરશો, dataset માં જવાબ નથી.'); qInput.value='';
});
qInput.addEventListener('keydown',e=>{if(e.key==='Enter') sendBtn.click();});
</script>
